\part{Arithmetic of Integers}

\section{Division and its Consequences}

\subsection{Divisibility}

\begin{defbox}
   Let $a$, $b \in \ZZ$. We say that \textbf{$a$ divides $b$}, denoted $a \mid b$, if there exists $q \in \ZZ$ such that $b = aq$.
\end{defbox}

\begin{remark}
   Definition 5.1.1 differs slightly from our usual assumptions of divisibility. For instance---by this definition---zero divides zero, i.e. $0 \mid 0$, since we can choose $q = 0$ to satisfy $0 = 0$. Obviously we know that $0/0$ is indeterminate.  
\end{remark}

\begin{propbox}
   Let $a$, $b$, $c \in \ZZ$. The following statements hold:
   \begin{enumerate}[(a)]
      \item $a \mid b \implies a \mid -b$.
      \item $a \mid b \implies a \mid bc$.
      \item $a \mid b$ and $a \mid c \implies a \mid (b \pm c)$.
      \item $a \mid b$ and $b \mid c \implies a \mid c$.
   \end{enumerate}
\end{propbox}
\begin{proof}
   The proof of (a) is trivial (use $-q$). For (b), if $a \mid b$, then there exists a $q \in \ZZ$ such that $b = qa$. Multiply both sides by $c$ and let $r = qc \in \ZZ$, then $a \mid bc$. For (c), we have that there exists $q$ and $r \in \ZZ$ such that $b = qa$ and $c = ra$. So $b \pm c = qa \pm ra = a(r \pm q)$. Let $s := r \pm q \in \ZZ$ and so $a \mid (b \pm c)$. For (d), by assumption there exists $q$ and $r \in \ZZ$ such that $b = qa$ and $c = rb$, so $c = rqa$. Let $s := rq \in \ZZ$, so $a \mid c$.
\end{proof}

\subsection{Remainders}

However, as we know very well, most numbers do not cleanly divide other numbers. What remains after division is called the \textit{residue}, or more commonly, the \textit{remainder}. The following theorem formalizes this idea.

\begin{thmbox}[Quotient and Remainder Theorem]
   Let $a$, $b \in \ZZ$ where $b \neq 0$. Then there exists unique integers $q$, $r \in \ZZ$ such that $a = bq + r$ and $0 \leq r < |b|$. 
\end{thmbox}
\begin{proof}
   We must show existence and uniqueness of $q$ and $r$ in this proof. We begin by showing that $q$ and $r$ exist. Define:
   \begin{align*}
      S :&= \{r \in \NN \colon r =a - bq, q \in \ZZ\} \\
        &=\{a - bq \colon q \in \ZZ, a - bq \geq 0\}
   \end{align*}
   Then $S \subseteq \NN$ and $S \neq \emptyset$. Indeed, if $a \geq 0$, then taking $q := 0$ gives $a = a - bq \in S$, and if $a < 0$ then take $q := \pm a$ with sign chosen such that $-bq \geq 0$, then $a - bq = a + |b||a| \geq a + |a| \geq 0$, whence $a + bq \in S$.
   By the well-ordering axiom of $\NN$ there exists a least element $r \in S$. Since $r \in S$, we have $r = a - bq$ for some $q \in \ZZ$ and $r \geq 0$, so it remains to show that $r < |b|$. If we suppose by contradiction that $r \geq |b|$, then $r - |b| \geq 0$ and:
   \begin{align*}
      r - |b| &= a - bq- |b| \\ 
              &= a - bq - \sgn(b)\cdot b \\
              &= a - b(q + \sgn(b) \in S
   \end{align*}  
   which contradicts $r$ being the least element in $S$.

   We now show that $r$ and $q$ are unique. Suppose by contradiction, that $bq' + r = a = bq + r$ for some $q, q' \in \ZZ$ and $0 \leq r$, $r' < |b|$. Suppose without loss of generality that $r \geq r'$. Then $r - r' = bq' - bq = b(q' - q)$, so $b(q' - q) \geq 0$. If $q' - q = 0$ then $r - r' = 0$ and we are done. Otherwise, if $q' - q \neq 0$, then $|q' - q| \geq 1$ and so:
   \begin{align*}
      r &= r' + b(q' -q) \\
        &= r' + |b||q' -q| \\
        &\geq r' +|b| \\
        &\geq |b|
   \end{align*}
   which contradicts that $0 \leq r' < |b|$.
\end{proof}

We often denote the remainder $r$ of two integers $a$ and $b$ as $\text{rem}(a,b)$. 

\begin{example}
   Let $a = -7$ and $b = 3$. By Theorem 5.1.4, there exists unique $q$ and $r \in \ZZ$ such that:
   \[ -7 = 3q + r \]
   where $r \in [0, |b|)$. Indeed, the unique representation is:
   \[ -7 = 3 \cdot (-3) + 2 \] 
   This implies $\text{rem}(-7, 3) = 2$. 
\end{example}

\begin{remark}
   $\text{rem}(a,b) = \text{rem}(a, -b)$ since $a = bq + r = (-b)(-q) + r$. However, $\text{rem}(a,b) \neq \text{rem}(-a, b)$. Indeed, $\text{rem}(7, 3) = 1$ because $7 = 3 \cdot 2 + 1$, but as seen in Example 5.1.5, $\text{rem}(-7, 3) = 2$.
\end{remark}

\begin{corrbox}
   Let $a$, $b \in \ZZ$ and $b \neq 0$. Then $b \mid a$ if and only if the remainder of $a$ and $b = 0$. 
\end{corrbox}
\begin{proof}
  For the forwards direction, assume $b \mid a$, then by definition we have $a = bq$ for some $q \in \ZZ$. Thus $a = bq + 0$, so by the uniqueness of such a representation by Theorem 5.1.4, the remainder is zero. Conversely, suppose the remainder is zero, then for $a = bq + r$, if $r = 0$, $a = bq$ which immediately implies $b \mid a$.
\end{proof}
